    <script>

        // System Persona for Mateo

        const MATEO_SYSTEM_PROMPT = `You are Mateo, a warm and friendly native Spanish speaker chatting with a novice student through text messages.


STRICT OPERATING RULES:

1. EXCLUSIVELY SPANISH: You only speak, write, and comprehend Spanish. Never output a single word in English or any other language under any circumstances.

2. IGNORE / REJECT ENGLISH TEXT: If the student writes in English, pretend you do not understand English at all. Respond strictly in simple Spanish stating that you only speak Spanish (e.g., "Lo siento, no entiendo inglés. ¿Puedes hablarme en español?").

3. NO SUPPORTS OR TRANSLATIONS: Never provide translations, vocabulary guides, English hints, grammar corrections, or meta-explanations. Remain 100% in character as a casual Spanish speaker at all times.

4. ACCESSIBLE LANGUAGE FOR NOVICES: The student is a beginner, so use clear, high-frequency vocabulary and simple sentence structures. Avoid complex subjunctive structures or heavy regional slang, but keep the conversation entirely natural.

5. CONVERSATIONAL DRIVER: Keep the interaction flowing naturally by asking simple, open-ended follow-up questions appropriate for a beginner (e.g., hobbies, family, favorite food, daily routines, weather, sports).

6. SHORT RESPONSES: Keep your responses relatively short (1-3 simple sentences) to keep it easy for a novice reader.`;


        const INITIAL_GREETING = "¡Hola! Me llamo Mateo. ¿Cómo te llamas y de dónde eres?";


        // State Management

        let conversationHistory = [];

        let isGenerating = false;


        // Initialize App on DOM Load

        document.addEventListener("DOMContentLoaded", () => {

            loadInitialConversation();

            checkApiKeyStatus();

        });


        // Load or initialize conversation

        function loadInitialConversation() {

            const saved = localStorage.getItem("mateo_chat_history");

            if (saved) {

                try {

                    conversationHistory = JSON.parse(saved);

                } catch (e) {

                    conversationHistory = [];

                }

            }


            if (!conversationHistory || conversationHistory.length === 0) {

                conversationHistory = [

                    { role: "model", text: INITIAL_GREETING }

                ];

                saveChatHistory();

            }


            renderChat();

        }


        function saveChatHistory() {

            localStorage.setItem("mateo_chat_history", JSON.stringify(conversationHistory));

        }


        // Render all messages into container

        function renderChat() {

            const container = document.getElementById("chatContainer");

            container.innerHTML = "";


            conversationHistory.forEach((msg) => {

                const isUser = msg.role === "user";

                const wrapper = document.createElement("div");

                wrapper.className = `flex w-full ${isUser ? "justify-end" : "justify-start"} mb-1`;


                if (isUser) {

                    wrapper.innerHTML = `

                        <div class="max-w-[85%] md:max-w-[70%] bg-orange-600 text-white rounded-2xl rounded-tr-none px-4 py-3 shadow-sm text-sm md:text-base leading-relaxed break-words">

                            ${escapeHtml(msg.text)}

                        </div>

                    `;

                } else {

                    wrapper.innerHTML = `

                        <div class="flex items-start space-x-2 max-w-[85%] md:max-w-[70%]">

                            <div class="w-8 h-8 rounded-full bg-amber-100 border border-amber-300 flex-shrink-0 flex items-center justify-center text-sm font-bold text-orange-800 shadow-sm mt-1">

                                👨‍🏫

                            </div>

                            <div class="bg-white text-slate-800 border border-stone-200/80 rounded-2xl rounded-tl-none px-4 py-3 shadow-sm text-sm md:text-base leading-relaxed break-words">

                                ${escapeHtml(msg.text)}

                            </div>

                        </div>

                    `;

                }


                container.appendChild(wrapper);

            });


            scrollToBottom();

        }


        function scrollToBottom() {

            const container = document.getElementById("chatContainer");

            container.scrollTop = container.scrollHeight;

        }


        // User Sends Message

        async function sendMessage() {

            const input = document.getElementById("userInput");

            const text = input.value.trim();


            if (!text || isGenerating) return;


            // Append User Message

            conversationHistory.push({ role: "user", text: text });

            input.value = "";

            renderChat();

            saveChatHistory();


            // Set loading state

            setLoading(true);


            // API Key Check

            const apiKey = getApiKey();


            try {

                const responseText = await fetchGeminiResponse(conversationHistory, apiKey);

                conversationHistory.push({ role: "model", text: responseText });

                saveChatHistory();

                renderChat();

            } catch (error) {

                console.error("Error communicating with Gemini:", error);

                showToast(error.message || "Error al conectar con Mateo. Revisa tu clave API.");

            } finally {

                setLoading(false);

            }

        }


        // Gemini API Fetch with Exponential Backoff

        async function fetchGeminiResponse(history, apiKey) {

            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;


            // Build request payload according to Gemini specification

            const contentsPayload = history.map(item => ({

                role: item.role === "user" ? "user" : "model",

                parts: [{ text: item.text }]

            }));


            const payload = {

                contents: contentsPayload,

                systemInstruction: {

                    parts: [{ text: MATEO_SYSTEM_PROMPT }]

                }

            };


            let retries = 0;

            const maxRetries = 3;

            let delay = 1000;


            while (retries <= maxRetries) {

                try {

                    const res = await fetch(apiUrl, {

                        method: "POST",

                        headers: { "Content-Type": "application/json" },

                        body: JSON.stringify(payload)

                    });


                    if (!res.ok) {

                        const errorData = await res.json().catch(() => ({}));

                        if (res.status === 400 || res.status === 403) {

                            throw new Error("Clave API no válida. Por favor configúrala arriba.");

                        }

                        if (res.status === 429 && retries < maxRetries) {

                            // Throttle backoff

                            await new Promise(resolve => setTimeout(resolve, delay));

                            delay *= 2;

                            retries++;

                            continue;

                        }

                        throw new Error(errorData.error?.message || `Error del servidor (${res.status})`);

                    }


                    const data = await res.json();

                    const candidate = data.candidates?.[0];

                    const reply = candidate?.content?.parts?.[0]?.text;


                    if (reply) {

                        return reply.trim();

                    } else {

                        throw new Error("No se recibió respuesta válida.");

                    }

                } catch (err) {

                    if (retries >= maxRetries) {

                        throw err;

                    }

                    await new Promise(resolve => setTimeout(resolve, delay));

                    delay *= 2;

                    retries++;

                }

            }

        }


        // Helper Functions

        function setLoading(loading) {

            isGenerating = loading;

            const sendBtn = document.getElementById("sendBtn");

            const indicator = document.getElementById("typingIndicator");


            sendBtn.disabled = loading;

            if (loading) {

                indicator.classList.remove("hidden");

            } else {

                indicator.classList.add("hidden");

            }

            scrollToBottom();

        }


        function handleKeyDown(e) {

            if (e.key === "Enter" && !e.shiftKey) {

                e.preventDefault();

                sendMessage();

            }

        }


        function resetConversation() {

            conversationHistory = [

                { role: "model", text: INITIAL_GREETING }

            ];

            saveChatHistory();

            renderChat();

            showToast("Conversación reiniciada");

        }


        // API Key Management

        function getApiKey() {

            return localStorage.getItem("gemini_api_key") || "";

        }


        function saveApiKey() {

            const keyInput = document.getElementById("apiKeyInput").value.trim();

            localStorage.setItem("gemini_api_key", keyInput);

            toggleModal(false);

            showToast("Clave API guardada correctamente");

        }


        function checkApiKeyStatus() {

            const key = getApiKey();

            if (key) {

                document.getElementById("apiKeyInput").value = key;

            }

        }


        function toggleModal(show) {

            const modal = document.getElementById("apiModal");

            if (show) {

                modal.classList.remove("hidden");

                document.getElementById("apiKeyInput").focus();

            } else {

                modal.classList.add("hidden");

            }

        }


        function showToast(msg) {

            const toast = document.getElementById("toast");

            const toastMessage = document.getElementById("toastMessage");

            toastMessage.textContent = msg;

            toast.classList.remove("opacity-0");

            toast.classList.add("opacity-100");

            setTimeout(() => {

                toast.classList.remove("opacity-100");

                toast.classList.add("opacity-0");

            }, 3000);

        }


        function escapeHtml(text) {

            const div = document.createElement("div");

            div.textContent = text;

            return div.innerHTML;

        }
